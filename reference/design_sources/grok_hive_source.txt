Switching to Rust, or a Rust/Python hybrid using PyO3, is an excellent choice for a high-throughput data factory.
Generating millions of synthetic data points requires massive network I/O (handling thousands of API requests concurrently) and heavy text processing (cleaning, regex filtering, and token tracking). Python’s Global Interpreter Lock (GIL) often bottlenecks this process. A Rust/PyO3 architecture gives you the best of both worlds: Rust handles the blazing-fast concurrent data pipeline, while Python handles data cleaning scripts or native machine learning calls.
Here is the architectural blueprint for your PyO3-powered synthetic data pipeline.

Why This Hybrid Setup Maximizes Performance

* Non-Blocking Async Requests: By utilizing Rust's tokio crate, you can fire off thousands of generation requests to local or remote LLM endpoints simultaneously. Python struggles with this at scale, but Rust handles it with negligible memory overhead.

* Targeted GIL Management: PyO3 allows your Rust code to explicitly acquire the Python GIL only when it needs to run a specialized Python utility (like an NLP cleaning library). The rest of your heavy pipeline runs natively in Rust at hardware speeds.

* Memory Safety at Scale: Processing 6 million total lines of text can cause memory leaks in pure Python. Rust guarantees memory safety and predictable memory allocation over long, multi-day generation runs.

```rust
use pyo3::prelude::*;
use pyo3::types::PyModule;
use serde::{Deserialize, Serialize};
use std::sync::Arc;

#[derive(Debug, Serialize, Deserialize, Clone)]
struct SeedChunk {
    id: String,
    content: String,
}

#[derive(Debug, Serialize, Deserialize)]
struct SyntheticOutput {
    vertical: String,
    prompt: String,
    thought_process: String,
    final_output: String,
}

// Embedded Python code for precise text cleaning via PyO3
const PYTHON_CLEANER: &str = r#"
import re

def clean_llm_text(text: str) -> str:
    # Aggressively strip out robotic AI tropes and filler phrases
    bad_phrases = [
        r"Here is a breakdown", 
        r"In conclusion", 
        r"It is important to remember", 
        r"Delve into"
    ]
    for pattern in bad_phrases:
        text = re.sub(pattern, "", text, flags=re.IGNORECASE)
    return text.strip()
"#;

async fn generate_synthetic_node(chunk: SeedChunk) -> Result<SyntheticOutput, Box<dyn std::error::Error + Send + Sync>> {
    let client = reqwest::Client::new();
    
    // 1. Fire off async request to your generation endpoint (e.g., local vLLM server or API)
    let response = client.post("http://localhost:8000/v1/chat/completions")
        .json(&serde_json::json!({
            "model": "llama-3.1-70b",
            "messages": [
                {"role": "system", "content": "You are a logical distillation engine. Synthesize a complex deductive reasoning puzzle based on this seed data."},
                {"role": "user", "content": chunk.content}
            ],
            "temperature": 0.7
        }))
        .send()
        .await?
        .json::<serde_json::Value>()
        .await?;

    // Extract raw text response safely
    let raw_generation = response["choices"][0]["message"]["content"]
        .as_str()
        .unwrap_or("")
        .to_string();

    // 2. Cross the PyO3 bridge to clean the text using the Python script
    let cleaned_generation = Python::with_gil(|py| -> PyResult<String> {
        let activator = PyModule::from_code_bound(py, PYTHON_CLEANER, "cleaner.py", "cleaner")?;
        let clean_fn = activator.getattr("clean_llm_text")?;
        let py_result = clean_fn.call1((raw_generation,))?;
        py_result.extract::<String>()
    })?;

    // 3. Map to clean structured JSON output
    Ok(SyntheticOutput {
        vertical: "deductive_reasoning".to_string(),
        prompt: format!("Solve the following puzzle derived from: {}", chunk.id),
        thought_process: "Parsed via pipeline".to_string(), // You would parse <thought> tags here
        final_output: cleaned_generation,
    })
}

#[tokio::main]
async fn main() {
    // Initialize Python interpreter inside the Rust process boundary
    pyo3::prepare_freethreaded_python();

    // Simulate an ingested batch of open-source data chunks
    let dataset_batch = vec![
        SeedChunk { id: "chunk_001".to_string(), content: "Raw data describing strict sequencing constraints...".to_string() },
        SeedChunk { id: "chunk_002".to_string(), content: "Raw legal case context displaying complex logical rule sets...".to_string() },
    ];

    let mut tasks = vec![];

    // Spawn concurrent tasks across all CPU threads via Tokio
    for chunk in dataset_batch {
        tasks.push(tokio::spawn(async move {
            match generate_synthetic_node(chunk).await {
                Ok(synthetic_data) => {
                    let json_line = serde_json::to_string(&synthetic_data).unwrap();
                    // In production, thread-safely append this line to your target .jsonl file
                    println!("Successfully Generated Row: {}", json_line);
                }
                Err(e) => eprintln!("Error generating row: {}", e),
            }
        }));
    }

    // Await all concurrent tasks to finish processing
    for task in tasks {
        let _ = task.await;
    }
}
```

  use futures::stream::{self, StreamExt};
  use pyo3::prelude::*;
  use pyo3::types::PyModule;
  use serde::{Deserialize, Serialize};
  use std::sync::Arc;
  use tokio::fs::OpenOptions;
  use tokio::io::AsyncWriteExt;
  use tokio::sync::Mutex;

  #[derive(Debug, Deserialize, Clone)]
  struct SeedChunk {
      id: String,
      content: String,
  }

  #[derive(Debug, Serialize)]
  struct SyntheticRecord {
      vertical: String,
      seed_id: String,
      generated_prompt: String,
      validator_thought: String,
      validated_output: String,
  }

  const PYTHON_CLEANER: &str = r#"
  import re
  def clean_and_verify(text: str) -> str:
      # Scrub out obvious AI boilerplate patterns
      text = re.sub(r"(?i)here is a breakdown|in conclusion|sure, let's look|delve into", "", text)
      return text.strip()
  "#;

  // Sends seed text to the local 70B Generator model
  async fn call_generator(client: &reqwest::Client, content: &str) -> Result<String, Box<dyn std::error::Error + Send + Sync>> {
      let res = client.post("http://localhost:8000/v1/chat/completions")
          .json(&serde_json::json!({
              "model": "casperhansen/llama-3.1-70b-instruct-awq",
              "messages": [
                  {"role": "system", "content": "Convert this source text into a complex, standalone multi-step deductive logic puzzle. Output ONLY the raw puzzle scenario, no filler."},
                  {"role": "user", "content": content}
              ],
              "temperature": 0.8
          }))
          .send()
          .await?
          .json::<serde_json::Value>()
          .await?;

      Ok(res["choices"][0]["message"]["content"].as_str().unwrap_or("").to_string())
  }

  // Sends the new puzzle to the local 8B Judge model to generate a reasoning path
  async fn call_validator(client: &reqwest::Client, puzzle: &str) -> Result<(String, String), Box<dyn std::error::Error + Send + Sync>> {
      let res = client.post("http://localhost:8001/v1/chat/completions")
          .json(&serde_json::json!({
              "model": "meta-llama/Llama-3.1-8B-Instruct",
              "messages": [
                  {"role": "system", "content": "Solve this puzzle step-by-step. You must think deeply. Format your thinking process inside <thought></thought> tags and put your short absolute final answer outside the tags."},
                  {"role": "user", "content": puzzle}
              ],
              "temperature": 0.2
          }))
          .send()
          .await?
          .json::<serde_json::Value>()
          .await?;

      let full_response = res["choices"][0]["message"]["content"].as_str().unwrap_or("").to_string();
      
      // Simple tag split logic
      let thought = full_response.split("</thought>").next().unwrap_or("").replace("<thought>", "").trim().to_string();
      let answer = full_response.split("</thought>").nth(1).unwrap_or(&full_response).trim().to_string();
      
      Ok((thought, answer))
  }

  #[tokio::main]
  async fn main() -> Result<(), Box<dyn std::error::Error>> {
      pyo3::prepare_freethreaded_python();
      let http_client = reqwest::Client::new();
      
      // Thread-safe wrapper for the output file
      let file = OpenOptions::new().create(true).append(true).open("deductive_reasoning_dataset.jsonl").await?;
      let shared_file = Arc::new(Mutex::new(file));

      // Mock incoming seed stream (in production, read a local JSONL file via tokio::io::BufReader)
      let raw_seeds = vec![
          SeedChunk { id: "chunk_101".to_string(), content: "Rule: Access requires clearances X and Y. John lacks Y.".to_string() },
          SeedChunk { id: "chunk_102".to_string(), content: "Sequence: Event A triggers B, but only if C is inactive.".to_string() },
      ];

      // Process up to 50 items concurrently using a managed futures buffer
      let concurrent_limit = 50;
      
      stream::iter(raw_seeds)
          .map(|seed| {
              let client = http_client.clone();
              let file_ref = shared_file.clone();
              async move {
                  // Step 1: Generate the puzzle
                  if let Ok(puzzle) = call_generator(&client, &seed.content).await {
                      
                      // Step 2: Use PyO3 to clean the prompt text instantly
                      let cleaned_puzzle = Python::with_gil(|py| -> PyResult<String> {
                          let module = PyModule::from_code_bound(py, PYTHON_CLEANER, "cleaner.py", "cleaner")?;
                          let clean_fn = module.getattr("clean_and_verify")?;
                          clean_fn.call1((puzzle,))?.extract::<String>()
                      }).unwrap_or(puzzle);

                      // Step 3: Run through Validator/Judge
                      if let Ok((thought, final_ans)) = call_validator(&client, &cleaned_puzzle).await {
                          if !final_ans.is_empty() && !thought.is_empty() {
                              
                              let record = SyntheticRecord {
                                  vertical: "deductive_reasoning".to_string(),
                                  seed_id: seed.id,
                                  generated_prompt: cleaned_puzzle,
                                  validator_thought: thought,
                                  validated_output: final_ans,
                              };

                              let serialized = serde_json::to_string(&record).unwrap() + "\n";
                              let mut locked_file = file_ref.lock().await;
                              let _ = locked_file.write_all(serialized.as_bytes()).await;
                          }
                      }
                  }
              }
          })
          .buffer_unordered(concurrent_limit)
          .collect::<()>()
          .await;

      println!("Batch generation step completed cleanly on DGX Spark.");
      Ok(())
  }

### Parallel Track: Mathematical Routing Mechanism for the Hive Orchestrator

While the fine-tune runs, we can design the router. Given your interest in deterministic systems and graph kernels (VEIN V2), here is a clean mathematical foundation that is still practical to implement in Rust.

**Core idea: “Suffering-Aware Graph Router”**

Model the hive as a directed graph:
- Nodes = specialized models (deductive-8b, math-8b, strategy-8b, …)
- Edge weight from A → B = `f( semantic_similarity(query, B) , historical_suffering_A_to_B , current_load_B )`

Where “suffering” is exactly the nociceptive metric we discussed earlier (context bloat + failed handoff rate + latency). The router solves a lightweight constrained optimization at inference time:

```
route(q) = argmin_B  [ α·(1 − sim(q, B)) + β·suffering(A→B) + γ·load(B) ]
subject to load(B) < capacity(B)
```

You can implement the similarity via cosine on embeddings from a tiny frozen encoder (or even the [CLS] token of the 8B itself). The suffering and load terms come from a small shared blackboard that every node writes to (exactly the `Arc<RwLock<Blackboard>>` pattern your Rust pipeline already uses).

This gives you:
- Deterministic fallback behavior (when suffering or load is high, route conservatively).
- Learnable parameters (α, β, γ) that can be tuned or even meta-learned later.
- Natural integration with your lifetime-bounded / epigenetic ideas — agents that cause high suffering simply get lower routing probability in the next generation.

### Design Summary (Suffering-Aware Graph Math)

- **Specialists** = the three nodes (Generator 70B on :8000, Validator 8B on :8001, Deductive-8B fine-tuned on :8002 after merge).
- **Blackboard** = shared `Arc<RwLock<Blackboard>>` that every call updates with:
  - `latency_ms`
  - `response_tokens` (proxy for context bloat)
  - `failure_count`
  - `last_used` timestamp
- **Semantic similarity** (first version) = simple but effective heuristic (Jaccard token overlap + length similarity). Easy to replace later with real embeddings from a tiny Candle model or a fourth vLLM embedding endpoint.
- **Cost function** (the mathematical core):
  ```rust
  cost = 0.45 * (1.0 - similarity)          // prefer semantic match
       + 0.35 * normalized_suffering        // penalize pain
       + 0.20 * normalized_load             // penalize hot nodes
  ```
- **Routing decision** = `argmin` over all specialists that are currently healthy.
- After every call we **update suffering** with an exponential moving average so the router learns in real time.

This gives you deterministic, auditable, low-latency routing that naturally avoids “painful” (slow, bloated, failing) paths — exactly the nociceptive idea we discussed earlier, now made practical.

### Full `router.rs`

```rust
// router.rs
// Suffering-aware graph router for the ARC-SWARM / Cede hive
// Drop-in module for your existing Rust/PyO3 data factory on DGX Spark

use reqwest::Client;
use serde::{Deserialize, Serialize};
use std::collections::HashMap;
use std::sync::Arc;
use std::time::{Duration, Instant};
use tokio::sync::RwLock;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SufferingMetrics {
    pub latency_ms: f32,
    pub response_tokens: usize,
    pub failure_count: u32,
    pub last_used: u64,           // unix timestamp
}

impl Default for SufferingMetrics {
    fn default() -> Self {
        Self {
            latency_ms: 0.0,
            response_tokens: 0,
            failure_count: 0,
            last_used: 0,
        }
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub enum Specialist {
    Generator70B,   // port 8000 - raw puzzle generation
    Validator8B,    // port 8001 - reasoning trace + final answer
    Deductive8B,    // port 8002 - fine-tuned deductive specialist (after merge)
}

impl Specialist {
    pub fn port(&self) -> u16 {
        match self {
            Specialist::Generator70B => 8000,
            Specialist::Validator8B => 8001,
            Specialist::Deductive8B => 8002,
        }
    }

    pub fn name(&self) -> &'static str {
        match self {
            Specialist::Generator70B => "generator-70b",
            Specialist::Validator8B => "validator-8b",
            Specialist::Deductive8B => "deductive-8b",
        }
    }
}

#[derive(Debug)]
pub struct Blackboard {
    pub metrics: HashMap<Specialist, SufferingMetrics>,
    pub total_calls: u64,
}

impl Blackboard {
    pub fn new() -> Self {
        let mut metrics = HashMap::new();
        for spec in [Specialist::Generator70B, Specialist::Validator8B, Specialist::Deductive8B] {
            metrics.insert(spec, SufferingMetrics::default());
        }
        Self { metrics, total_calls: 0 }
    }
}

pub type SharedBlackboard = Arc<RwLock<Blackboard>>;

/// Simple but effective semantic similarity (replace with real embeddings later)
fn semantic_similarity(query: &str, specialist: Specialist) -> f32 {
    // Heuristic: token overlap + length similarity
    let q_tokens: Vec<&str> = query.split_whitespace().collect();
    let specialist_bias = match specialist {
        Specialist::Generator70B => 0.6,   // good at creative puzzle generation
        Specialist::Validator8B => 0.85,   // excellent at structured reasoning
        Specialist::Deductive8B => 0.95,   // best at pure deductive logic (fine-tuned)
    };

    let overlap = q_tokens.len() as f32 / (q_tokens.len() as f32 + 5.0); // crude
    (overlap * 0.6 + specialist_bias * 0.4).clamp(0.0, 1.0)
}

/// Core suffering-aware cost function
fn compute_cost(similarity: f32, suffering: &SufferingMetrics, load: f32) -> f32 {
    let norm_suffering = (suffering.latency_ms / 5000.0 + suffering.failure_count as f32 * 0.1)
        .min(1.0);
    let norm_load = load.min(1.0);

    0.45 * (1.0 - similarity) + 0.35 * norm_suffering + 0.20 * norm_load
}

/// Main routing + calling function
pub async fn route_and_call(
    query: &str,
    blackboard: &SharedBlackboard,
    client: &Client,
) -> Result<String, Box<dyn std::error::Error + Send + Sync>> {
    let start = Instant::now();

    // 1. Read current suffering + load
    let (best_specialist, _cost) = {
        let board = blackboard.read().await;
        let mut best = Specialist::Deductive8B;
        let mut best_cost = f32::MAX;

        for &spec in &[
            Specialist::Generator70B,
            Specialist::Validator8B,
            Specialist::Deductive8B,
        ] {
            let metrics = board.metrics.get(&spec).unwrap();
            let load = (board.total_calls as f32 / 1000.0).min(1.0); // simple load proxy
            let sim = semantic_similarity(query, spec);
            let cost = compute_cost(sim, metrics, load);

            if cost < best_cost {
                best_cost = cost;
                best = spec;
            }
        }
        (best, best_cost)
    };

    // 2. Call the chosen specialist
    let url = format!("http://localhost:{}/v1/chat/completions", best_specialist.port());
    let payload = serde_json::json!({
        "model": best_specialist.name(),
        "messages": [
            {"role": "system", "content": "You are a precise deductive reasoning engine."},
            {"role": "user", "content": query}
        ],
        "temperature": if best_specialist == Specialist::Deductive8B { 0.2 } else { 0.7 },
        "max_tokens": 2048
    });

    let resp = client.post(&url).json(&payload).send().await?;
    let latency = start.elapsed().as_millis() as f32;

    if !resp.status().is_success() {
        // Update failure
        let mut board = blackboard.write().await;
        if let Some(m) = board.metrics.get_mut(&best_specialist) {
            m.failure_count += 1;
        }
        return Err(format!("{} returned status {}", best_specialist.name(), resp.status()).into());
    }

    let json: serde_json::Value = resp.json().await?;
    let output = json["choices"][0]["message"]["content"]
        .as_str()
        .unwrap_or("")
        .to_string();

    // 3. Update blackboard with real suffering signal
    {
        let mut board = blackboard.write().await;
        if let Some(m) = board.metrics.get_mut(&best_specialist) {
            // Exponential moving average
            let alpha = 0.3;
            m.latency_ms = alpha * latency + (1.0 - alpha) * m.latency_ms;
            m.response_tokens = output.split_whitespace().count();
            m.last_used = std::time::SystemTime::now()
                .duration_since(std::time::UNIX_EPOCH)
                .unwrap()
                .as_secs();
        }
        board.total_calls += 1;
    }

    Ok(output)
}
```

            Specialist::Validator8B => "validator-8b",
            Specialist::Deductive8B => "deductive-8b",
        }
    }
}

#[derive(Debug, Serialize, Deserialize)]
pub struct Blackboard {
    pub metrics: HashMap<Specialist, SufferingMetrics>,
    pub total_calls: u64,
}

impl Blackboard {
    pub async fn new() -> Self {
        if let Ok(data) = fs::read_to_string("blackboard.json").await {
            if let Ok(board) = serde_json::from_str::<Blackboard>(&data) {
                println!("✅ Loaded previous suffering state from blackboard.json");
                return board;
            }
        }
        let mut metrics = HashMap::new();
        for spec in [Specialist::Generator70B, Specialist::Validator8B, Specialist::Deductive8B] {
            metrics.insert(spec, SufferingMetrics::default());
        }
        Self { metrics, total_calls: 0 }
    }

    async fn save_to_disk(&self) {
        if let Ok(json) = serde_json::to_string_pretty(self) {
            let _ = fs::write("blackboard.json", json).await;
        }
    }
}

pub type SharedBlackboard = Arc<RwLock<Blackboard>>;

fn semantic_similarity(query: &str, specialist: Specialist) -> f32 {
    let q_tokens: Vec<&str> = query.split_whitespace().collect();
    let specialist_bias = match specialist {
        Specialist::Generator70B => 0.60,
        Specialist::Validator8B => 0.85,
        Specialist::Deductive8B => 0.95,
    };
    let overlap = q_tokens.len() as f32 / (q_tokens.len() as f32 + 5.0);
    (overlap * 0.6 + specialist_bias * 0.4).clamp(0.0, 1.0)
}

fn compute_cost(similarity: f32, suffering: &SufferingMetrics, load: f32) -> f32 {
    let norm_suffering = (suffering.latency_ms / 5000.0 + suffering.failure_count as f32 * 0.1).min(1.0);
    let norm_load = load.min(1.0);
    0.45 * (1.0 - similarity) + 0.35 * norm_suffering + 0.20 * norm_load
}

pub async fn route_and_call(
    query: &str,
    blackboard: &SharedBlackboard,
    client: &Client,
) -> Result<String, Box<dyn std::error::Error + Send + Sync>> {
    let start = Instant::now();

    let best_specialist = {
        let board = blackboard.read().await;
        let mut best = Specialist::Deductive8B;
        let mut best_cost = f32::MAX;

        for &spec in &[Specialist::Generator70B, Specialist::Validator8B, Specialist::Deductive8B] {
            let metrics = board.metrics.get(&spec).unwrap();
            let load = (board.total_calls as f32 / 1000.0).min(1.0);
            let sim = semantic_similarity(query, spec);
            let cost = compute_cost(sim, metrics, load);
            if cost < best_cost {
                best_cost = cost;
                best = spec;
            }
        }
        best
    };

    let url = format!("http://localhost:{}/v1/chat/completions", best_specialist.port());
    let payload = serde_json::json!({
        "model": best_specialist.name(),
        "messages": [
            {"role": "system", "content": "You are a precise deductive reasoning engine."},
            {"role": "user", "content": query}
        ],
        "temperature": if best_specialist == Specialist::Deductive8B { 0.2 } else { 0.7 },
        "max_tokens": 2048
    });

    let resp = client.post(&url).json(&payload).send().await?;
    let latency = start.elapsed().as_millis() as f32;

    let output = if resp.status().is_success() {
        let json: serde_json::Value = resp.json().await?;
        json["choices"][0]["message"]["content"].as_str().unwrap_or("").to_string()
    } else {
        let mut board = blackboard.write().await;
        if let Some(m) = board.metrics.get_mut(&best_specialist) {
            m.failure_count += 1;
        }
        tokio::spawn(async move {
            let board = blackboard.read().await;
            board.save_to_disk().await;
        });
        return Err(format!("{} failed with status {}", best_specialist.name(), resp.status()).into());
    };

    // Update suffering + persist
    {
        let mut board = blackboard.write().await;
        if let Some(m) = board.metrics.get_mut(&best_specialist) {
            let alpha = 0.3;
            m.latency_ms = alpha * latency + (1.0 - alpha) * m.latency_ms;
            m.response_tokens = output.split_whitespace().count();
            m.last_used = std::time::SystemTime::now()
                .duration_since(std::time::UNIX_EPOCH)
                .unwrap()
                .as_secs();
        }
        board.total_calls += 1;
        let board_clone = board.clone(); // cheap because HashMap is small
        tokio::spawn(async move {
            board_clone.save_to_disk().await;
        });
    }

    Ok(output)
}

            total_calls = 0;

            for line in body.lines() {
                if line.starts_with("hive_specialist_latency_ms") {
                    if let Some((name, value)) = parse_metric(line) {
                        metrics.entry(name).or_default().latency_ms = value;
                    }
                } else if line.starts_with("hive_specialist_response_tokens") {
                    if let Some((name, value)) = parse_metric(line) {
                        metrics.entry(name).or_default().response_tokens = value as usize;
                    }
                } else if line.starts_with("hive_specialist_failures") {
                    if let Some((name, value)) = parse_metric(line) {
                        metrics.entry(name).or_default().failures = value as u32;
                    }
                } else if line.starts_with("hive_total_calls") {
                    if let Some(value) = line.split_whitespace().last() {
                        total_calls = value.parse().unwrap_or(0);
                    }
                }
            }
            last_update = Instant::now();
        }

        terminal.draw(|f| {
            let chunks = Layout::default()
                .direction(Direction::Vertical)
                .constraints([
                    Constraint::Min(12),
                    Constraint::Length(6),
                    Constraint::Length(3),
                ])
                .split(f.size());

            // === Latency Bar Chart ===
            let mut data: Vec<(&str, u64)> = metrics
                .iter()
                .map(|(name, m)| (name.as_str(), m.latency_ms as u64))
                .collect();
            data.sort_by_key(|(_, v)| *v);

            let barchart = BarChart::default()
                .block(Block::default().title("Specialist Latency (ms)").borders(Borders::ALL))
                .data(&data)
                .bar_width(14)
                .bar_style(Style::default().fg(Color::Cyan))
                .value_style(Style::default().fg(Color::White));

            f.render_widget(barchart, chunks[0]);

            // === Status Table ===
            let mut status_lines = String::new();
            for (name, m) in &metrics {
                let color = if m.latency_ms > 800.0 || m.failures > 5 {
                    "🔴"
                } else if m.latency_ms > 300.0 {
                    "🟡"
                } else {
                    "🟢"
                };
                status_lines.push_str(&format!(
                    "{} {:<14} | Latency: {:>6.1}ms | Tokens: {:>5} | Failures: {}\n",
                    color, name, m.latency_ms, m.response_tokens, m.failures
                ));
            }

            let status = Paragraph::new(status_lines)
                .block(Block::default().title("Specialist Status").borders(Borders::ALL));
            f.render_widget(status, chunks[1]);

            // === Footer ===
            let footer = Paragraph::new(format!(
                "Total Calls: {}   |   Last Update: {:.1}s ago   |   Press 'q' to quit & stop hive",
                total_calls,
                last_update.elapsed().as_secs_f32()
            ))
            .block(Block::default().title("ARC-SWARM / Cede Hive Dashboard").borders(Borders::ALL));
            f.render_widget(footer, chunks[2]);
        })?;

        // Keyboard input
        if event::poll(Duration::from_millis(150))? {
            if let Event::Key(key) = event::read()? {
                if key.code == KeyCode::Char('q') {
                    should_quit = true;
                }
            }
        }

        tokio::time::sleep(Duration::from_millis(650)).await;
    }

    disable_raw_mode()?;
    io::stdout().execute(LeaveAlternateScreen)?;
    Ok(())
}

/// Very small Prometheus text format parser
fn parse_metric(line: &str) -> Option<(String, f64)> {
    let parts: Vec<&str> = line.split_whitespace().collect();
    if parts.len() < 2 {
        return None;
    }

    // Extract specialist name from labels
    let name = if let Some(start) = line.find("specialist=\"") {
        let rest = &line[start + 12..];
        if let Some(end) = rest.find('"') {
            rest[..end].to_string()
        } else {
            "unknown".to_string()
        }
    } else {
        "unknown".to_string()
    };

    if let Ok(value) = parts.last()?.parse::<f64>() {
        Some((name, value))
    } else {
        None
    }
}
