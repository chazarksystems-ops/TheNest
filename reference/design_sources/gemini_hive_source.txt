### 1. The Immutable Auditing Ledger (The Graveyard)

Every time an agent completes a task or undergoes apoptosis, we serialize its `EpigeneticPayload` to an append-only cryptographic ledger. ```rust
use serde::{Serialize, Deserialize};
use bincode;
use blake3::Hash;
use tokio::fs::OpenOptions;
use tokio::io::AsyncWriteExt;

#[derive(Serialize, Deserialize, Debug, Clone)]
pub struct LedgedEvent {
    pub timestamp: u64,
    pub previous_hash: Hash,
    pub payload: EpigeneticPayload,
}

pub struct ImmutableLedger {
    file_path: String,
    last_hash: Hash,
}

impl ImmutableLedger {
    pub fn new(path: &str) -> Self {
        // Initialize with a genesis hash
        ImmutableLedger {
            file_path: path.to_string(),
            last_hash: blake3::hash(b"GENESIS_BLOCK"),
        }
    }

    /// Appends the death rattle to the cryptographic graveyard.
    pub async fn record_apoptosis(&mut self, payload: EpigeneticPayload) -> Result<(), std::io::Error> {
        let event = LedgedEvent {
            timestamp: std::time::SystemTime::now()
                .duration_since(std::time::UNIX_EPOCH).unwrap().as_secs(),
            previous_hash: self.last_hash,
            payload,
        };

        // Serialize directly to binary bytes
        let encoded: Vec<u8> = bincode::serialize(&event).unwrap();
        
        // Cryptographically chain the next hash
        self.last_hash = blake3::hash(&encoded);

        // Append-only write to disk
        let mut file = OpenOptions::new()
            .create(true)
            .append(true)
            .open(&self.file_path)
            .await?;

        file.write_all(&encoded).await?;
        file.sync_data().await?;

        println!("Apoptosis cryptographically sealed in ledger. Hash: {}", self.last_hash.to_hex());
        Ok(())
    }
}
